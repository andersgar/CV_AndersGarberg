  \vspace{\entrysep}\item
    \begin{tabular*}{\textwidth}[t]{l@{\extracolsep{\fill}}r}

    \end{tabular*}\vspace{\hugup}

  \vspace{\entrysep}\item

      {\color{dark-grey}\small #3} & \\
    \end{tabular*}\vspace{\hugup}
}

}
